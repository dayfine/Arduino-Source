cmake_minimum_required(VERSION 3.24)
project(SourceGroupFixture NONE)
include("${SOURCE_GROUPS_MODULE}")
set(SOURCES)
if(SCENARIO STREQUAL "mixed" OR SCENARIO STREQUAL "authored")
    list(APPEND SOURCES CMakeLists.txt)
endif()
if(SCENARIO STREQUAL "mixed" OR SCENARIO STREQUAL "all_generated")
    # qt_add_resources() marks its binary-directory qrc output GENERATED.
    set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/qrc_darkstyle.cpp")
    set_source_files_properties("${OUTPUT}" PROPERTIES GENERATED TRUE)
    list(APPEND SOURCES "${OUTPUT}")
endif()
if(SCENARIO STREQUAL "invalid")
    set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/unrelated.cpp")
    file(WRITE "${OUTPUT}" "// Authored outside the source tree.\n")
    list(APPEND SOURCES "${OUTPUT}")
endif()
pa_source_groups("${CMAKE_CURRENT_SOURCE_DIR}" ${SOURCES})
